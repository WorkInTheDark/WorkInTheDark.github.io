<script>
/* Client-side filtering for the episode listing. Every card is rendered by
 * Jekyll above; this only hides the ones that do not match the active chapter
 * chip, the "new since" selector, and the search box, so the page degrades to a
 * full listing without JS.
 *
 * The "my last visit" option works off two localStorage keys: `last-visit` is
 * rolled forward to today on every load, and the value it held on an earlier day
 * moves to `prev-visit`. Reading `prev-visit` therefore answers "what had I not
 * seen when I arrived today", and it survives several visits on the same day. */
(function () {
  var chips = Array.prototype.slice.call(document.querySelectorAll('.phd101-chip'));
  var search = document.getElementById('phd101-search');
  var sinceGroup = document.querySelector('.phd101-since');
  var sinceSelect = document.getElementById('phd101-since-select');
  var sinceDate = document.getElementById('phd101-since-date');
  var sections = Array.prototype.slice.call(document.querySelectorAll('.phd101-chapter-group'));
  var countEl = document.querySelector('.phd101-result-count');
  var emptyEl = document.querySelector('.phd101-empty');
  if (!chips.length || !sections.length) return;

  var total = {{ total }};
  var state = { chapter: 'all', query: '', since: '' };

  var LAST_KEY = 'phd101:last-visit';
  var PREV_KEY = 'phd101:prev-visit';

  function iso(date) {
    var month = String(date.getMonth() + 1);
    var day = String(date.getDate());
    return date.getFullYear() + '-' +
      (month.length < 2 ? '0' + month : month) + '-' +
      (day.length < 2 ? '0' + day : day);
  }

  /* Parsed as local time on purpose: `new Date('2026-08-10')` is UTC midnight,
   * which formats as the previous day west of Greenwich. */
  function pretty(isoDate) {
    var parts = isoDate.split('-');
    var date = new Date(+parts[0], +parts[1] - 1, +parts[2]);
    return date.toLocaleDateString(undefined, { day: 'numeric', month: 'short', year: 'numeric' });
  }

  function daysAgo(days) {
    var date = new Date();
    date.setDate(date.getDate() - days);
    return iso(date);
  }

  /* Roll the visit stamps forward, and hand back the previous visit (if any). */
  function trackVisit() {
    var today = iso(new Date());
    var previous = null;
    try {
      var last = window.localStorage.getItem(LAST_KEY);
      previous = window.localStorage.getItem(PREV_KEY);
      if (last && last !== today) {
        previous = last;
        window.localStorage.setItem(PREV_KEY, previous);
      }
      window.localStorage.setItem(LAST_KEY, today);
    } catch (error) {
      previous = null;   /* private browsing, or storage disabled */
    }
    return previous;
  }

  var lastVisit = trackVisit();

  function cutoff() {
    if (state.since === 'visit') return lastVisit;
    if (state.since === 'custom') return sinceDate.value || null;
    if (state.since && state.since !== 'all') return daysAgo(parseInt(state.since, 10));
    return null;
  }

  function apply() {
    var since = cutoff();
    var shown = 0;

    sections.forEach(function (section) {
      var visibleInSection = 0;
      var cards = section.querySelectorAll('.phd101-card');

      Array.prototype.forEach.call(cards, function (card) {
        var matchesChapter =
          state.chapter === 'all' || card.getAttribute('data-chapter') === state.chapter;
        var matchesQuery =
          !state.query || card.getAttribute('data-search').indexOf(state.query) > -1;
        /* ISO dates compare correctly as strings. */
        var matchesDate = !since || card.getAttribute('data-date') >= since;
        var visible = matchesChapter && matchesQuery && matchesDate;

        card.hidden = !visible;
        if (visible) visibleInSection++;
      });

      section.hidden = visibleInSection === 0;
      shown += visibleInSection;
    });

    emptyEl.textContent = since && !state.query
      ? 'No new episodes since ' + pretty(since) + '.'
      : 'No episode matches that search.';
    emptyEl.hidden = shown > 0;

    if (state.chapter === 'all' && !state.query && !since) {
      countEl.textContent = '';
    } else if (since) {
      countEl.textContent = shown + ' of ' + total + ' episodes, published since ' + pretty(since);
    } else {
      countEl.textContent = shown + ' of ' + total + ' episodes';
    }
  }

  chips.forEach(function (chip) {
    chip.addEventListener('click', function () {
      state.chapter = chip.getAttribute('data-chapter');
      chips.forEach(function (other) {
        var active = other === chip;
        other.classList.toggle('is-active', active);
        other.setAttribute('aria-pressed', active ? 'true' : 'false');
      });
      apply();
    });
  });

  search.addEventListener('input', function () {
    state.query = search.value.trim().toLowerCase();
    apply();
  });

  if (sinceGroup && sinceSelect && sinceDate) {
    /* Nothing here does anything without JS, so it stays hidden until now. */
    sinceGroup.hidden = false;

    if (!lastVisit) {
      var visitOption = sinceSelect.querySelector('option[value="visit"]');
      visitOption.disabled = true;
      visitOption.textContent = 'my last visit (first visit)';
    }

    sinceSelect.addEventListener('change', function () {
      state.since = sinceSelect.value === 'all' ? '' : sinceSelect.value;
      sinceDate.hidden = sinceSelect.value !== 'custom';
      if (sinceSelect.value === 'custom' && !sinceDate.value) {
        sinceDate.focus();
        return;   /* wait for a date before filtering to nothing */
      }
      apply();
    });

    sinceDate.addEventListener('change', apply);
  }

  apply();
})();
</script>
